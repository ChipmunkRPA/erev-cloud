// Supplementary screen captures for the illustrated guide (docs/guides/illustrated-guide.md): the routed
// screens of Settings and of sign-in that the `screens` project does not capture. This file is NOT part of the
// release candidate's e2e projects: it was written on 2026-10-04 and run once, on the seeded open world, under
// the `screens` project's file pattern (`make e2e PROJECT=screens SPEC=extra-screens.spec.ts`), with the same
// capture helper (frontend/e2e/support/screens.ts). It opens each screen as a demo persona, waits for its
// heading and its reads, and captures both themes; it asserts nothing else and changes no data.
import type { Page } from "@playwright/test";

import { personaEmail } from "../support/auth";
import { expect, test } from "../support/fixtures";

interface Listed<T> {
  readonly items: readonly T[];
}

async function listed<T>(
  page: Page,
  path: string,
  params: Record<string, string> = {},
): Promise<readonly T[]> {
  const response = await page.request.get(path, { params });
  expect(response.ok(), `${path} answered ${response.status()}`).toBe(true);
  return ((await response.json()) as Listed<T>).items;
}

/** The page has drawn its heading and its reads have come. */
async function settled(page: Page, title?: string): Promise<void> {
  const heading =
    title === undefined
      ? page.getByRole("heading", { level: 1 }).first()
      : page.getByRole("heading", { level: 1, name: title, exact: true });
  await expect(heading).toBeVisible();
  await page.waitForLoadState("networkidle");
  await page.waitForTimeout(500);
}

  ["sf-15-calendars", "SF-15:calendars", "/settings/calendars", "Calendars"],
  ["sf-15-currencies", "SF-15:currencies", "/settings/currencies", "Currencies and rates"],
  ["sf-15-customers", "SF-15:customers", "/settings/customers", "Customers"],
  [
    "sf-15-related-party-groups",
    "SF-15:related-party-groups",
    "/settings/related-party-groups",
    "Related-party groups",
  ],
  ["sf-15-products", "SF-15:products", "/settings/products", "Products"],
  ["sf-14-users", "SF-14", "/settings/users", "Users"],
  ["sf-14-roles", "SF-14:roles", "/settings/roles", "Roles"],
  ["sf-14-sod", "SF-14:sod", "/settings/separation-of-duties", "Separation of duties"],
  ["sf-14-access-reviews", "SF-14:access-reviews", "/settings/access-reviews", "Access reviews"],

  ["sf-16-developer", "SF-16:developer", "/settings/developer", "API clients and webhooks"],
] as const;

for (const [name, surface, path, title] of TOMAS_PAGES) {
  test.describe(`${surface} ${title} (${path}), persona tomas`, () => {
    test(`${surface} as tomas`, async ({ personas, screens }) => {
      const page = await personas.page("tomas");
      await page.goto(path);
      await settled(page, title);
      await screens.capture(page, name, { surface });
    });
  });
}

test.describe("SF-15:customer Customer (/settings/customers/:customerId), persona tomas", () => {
  test("SF-15:customer as tomas", async ({ personas, screens }) => {
    const page = await personas.page("tomas");
    const customers = await listed<{ readonly id: string; readonly name: string }>(
      page,
      "/api/v1/customers",
      { q: "Pellworth" },
    );
    const customer = customers[0];
    expect(customer, "a customer named Pellworth").toBeDefined();
    await page.goto(`/settings/customers/${customer?.id ?? ""}`);
    await settled(page);
    await screens.capture(page, "sf-15-customer", { surface: "SF-15:customer" });
  });
});
